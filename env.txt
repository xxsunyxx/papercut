# ---- 后端服务器设置 ----
# 监听地址:
#   0.0.0.0 = 监听所有网卡，局域网内其他设备可通过 IP 访问（推荐）
#   127.0.0.1 = 仅本机可访问
HOST=0.0.0.0
# 后端 API 监听端口
PORT=8000

# ---- 前端服务器设置 ----
# 前端 Web UI 监听地址
FRONTEND_HOST=0.0.0.0
# 前端 Web UI 监听端口
FRONTEND_PORT=987
# 前端代理后端 API 的目标地址
FRONTEND_API_TARGET=http://localhost:8000
# 允许访问前端的主机名（逗号分隔，* 表示允许所有）
#   如果你使用内网穿透或域名访问，需在此添加对应域名
FRONTEND_ALLOWED_HOSTS=hom.my.to,*
#   示例: FRONTEND_ALLOWED_HOSTS=localhost,127.0.0.1,hom.my.to

# ---- PaddleOCR 模型设置 ----
# 推理引擎:
#   paddle_dynamic = 动态图引擎（推荐，兼容性最好）
#   paddle = 静态图引擎
#   onnxruntime = ONNX 引擎（需额外安装 onnxruntime）
OCR_ENGINE=paddle_dynamic
# 文字检测模型名称（用于定位文字位置，不识别内容）
#   PP-OCRv5_mobile_det = 移动版，速度快（推荐）
#   PP-OCRv5_server_det = 服务器版，精度高但慢
OCR_DET_MODEL=PP-OCRv5_mobile_det
# 文字识别模型名称（当前仅检测模式不使用，保留配置）
OCR_REC_MODEL=PP-OCRv5_mobile_rec
# 识别语言: ch=中文, en=英文, japan=日文, korean=韩文
OCR_LANG=ch

# ---- 模型存储 ----
# 模型文件存放目录（相对于 webui 根目录）
# 首次运行时自动下载到此目录，之后使用缓存
# 迁移时将此目录一起复制即可，无需重新下载
PADDLEX_HOME=models

# ---- 图像处理 ----
# OCR 分析时的图像缩放比例
#   1.0 = 原始分辨率（精度最高，速度较慢）
#   0.5 = 缩小一半（速度提升约 2 倍，可能漏检小字）
OCR_IMAGE_SCALE=1.0
# PDF 渲染倍率（值越大清晰度越高，但渲染和检测越慢）
#   2.0 = 2倍渲染（推荐，A3试卷清晰度足够）
#   1.5 = 1.5倍渲染（更快，但小字可能模糊）
PDF_RENDER_SCALE=2.0

# ---- 密封线检测 ----
# 密封线识别的最小覆盖率
#   垂直线覆盖页面高度的比例需超过此值才判定为密封线
#   0.35 = 35%（即虚线长度需达到页面高度的 35% 以上）
#   调低（如 0.20）= 更灵敏，可能误检
#   调高（如 0.50）= 更严格，可能漏检
SEAL_LINE_MIN_COVERAGE=0.15
# 密封线搜索区域宽度（占页面宽度的比例）
#   0.20 = 分别搜索页面左侧 20% 和右侧 20% 区域
#   （试卷正面密封线在左侧，背面翻面后在右侧）
#   如果密封线位置偏中间，可适当增大此值
SEAL_LINE_SEARCH_WIDTH=0.20

# ---- 性能优化 ----
# 禁用 OneDNN（Intel 加速库），解决部分 CPU 兼容性问题
#   0 = 禁用（推荐）
#   1 = 启用
FLAGS_USE_MKLDNN=1
# OpenMP 线程数：每个模型实例内部的 OpenMP 线程数
# 建议保持 1，通过多模型池实现真正并行
OMP_NUM_THREADS=1
# 页面并行处理线程数（每页一个线程，真正的并发加速）
#   0 = 自动（按 CPU 核心数决定，不再限制上限）
#   1 = 串行处理（关闭多线程）
#   4 = 4 页同时处理
PAGE_WORKERS=0
# 模型池大小：每个 worker 持有独立 OCR 实例，真正并行
#   0 = 自动（取 PAGE_WORKERS 或 CPU核心数/2，上限 16）
#   4 = 创建 4 个独立 OCR 模型实例
MODEL_POOL_SIZE=0
